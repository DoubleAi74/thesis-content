\subsection*{Quick-reference specialisations}

\paragraph{$\mu=0$.} $b=0$, $a=1+\delta/\lambda$, $D\equiv0$, $\Ifix=I$,
$V_\infty=1+\lambda/\delta=\langle X\rangle_{\rm QS}$,
$\mean{T_{\rm prod}}=\lambda^{-1}\log(1+\lambda/\delta)$,
$\pr{\Kb=k}=s\rho^{k-1}$ with $s=\delta/(\lambda+\delta)$, $\rho=1-s$;
flooding favours bursting for establishment at every $\delta>0$.

\paragraph{The boundary $L=1$.} $\delta=\lambda\mu/(\lambda+\mu)$,
$b=\delta/\lambda$; the offspring law is geometric with mean $m$;
$z_{\rm ext}^{\rm burst}=z_{\rm ext}^{\rm bud}$ for every $q$; offspring
variances equal.

\paragraph{Young-cell, fast-growth limit.} $p_{\rm eff}\to\delta$,
$d_{\rm eff}\to\mu+\delta$: a newly infected cell holds one particle, bursts
at rate $\delta$, releases one.

\paragraph{Set-point, $r=0$ matching.}
$p_{\rm eff}=V_\infty/\mean{T_{\rm prod}}$,
$d_{\rm eff}=1/\mean{T_{\rm prod}}$: lifetime output over lifetime; the
fairest constant-parameter comparison, and the one used in
\cref{p:sec:overlays}.
