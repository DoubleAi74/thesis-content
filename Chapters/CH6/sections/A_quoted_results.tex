\section{Results quoted from earlier chapters}
\label{p:app:quoted}

The body uses the single-cell closed forms of \ChCore{} constantly and the
distribution-level facts of \ChDist{} at isolated points. This appendix
quotes both, so that the introduction need not reprint them.
\Cref{p:fig:handoff} records what is handed across.

\subsection{Roots, identities and moments}

Write $w=\ee^{\theta t}$. The roots of the quadratic
$\lambda f^2-(\lambda+\mu+\delta)f+\mu$ are
\begin{equation}
a=\eta+\sqrt{\eta^2-\frac{\mu}{\lambda}},\qquad
b=\eta-\sqrt{\eta^2-\frac{\mu}{\lambda}},\qquad
\eta=\frac{\lambda+\mu+\delta}{2\lambda},
\label{p:eq:roots}
\end{equation}
with $b<1<a$, and the derived constants are
\begin{equation}
A=a-1,\qquad B=1-b,\qquad \theta=\lambda(a-b),\qquad \kappa=1+\frac{\delta}{2\lambda}.
\label{p:eq:derived}
\end{equation}
The identities that simplify everything downstream are
\begin{equation}
ab=\frac{\mu}{\lambda},\quad a+b=\frac{\lambda+\mu+\delta}{\lambda},\quad
AB=\frac{\delta}{\lambda},\quad A+B=a-b,\quad \theta=\lambda(a-b).
\label{p:eq:AB}
\end{equation}
The no-burst probability, the internal-extinction probability, and the
non-fixation probability are
\begin{equation}
I(t)=\frac{aB+bAw}{B+Aw},\qquad
D(t)=\frac{ab\,(w-1)}{aw-b},\qquad
\Ifix(t)=I(t)-D(t)=\frac{(a-b)^2\,w}{(B+Aw)(aw-b)}.
\label{p:eq:IDIhat}
\end{equation}
Here $I(\infty)=D(\infty)=b$, the probability that the cell clears
internally without ever bursting, while $\Ifix(0)=1$,
$\Ifix'(0)=-(\mu+\delta)$ and $\Ifix(\infty)=0$. With $I'=-\delta J$ and
$V'=\delta K$,
\begin{equation}
J(t)=\frac{(a-b)^2\,w}{(B+Aw)^2}=-\frac{\lambda}{\delta}(I-a)(I-b),
\label{p:eq:J}
\end{equation}
\begin{equation}
K(t)=\Bigl[1+\frac{2\lambda}{\delta}(1-I)\Bigr]J
=\frac{2\lambda}{\delta}(\kappa-I)J,
\label{p:eq:K}
\end{equation}
and the cumulative mean release is a quadratic in $1-I$ alone,
\begin{equation}
V(t)=(1-I)\Bigl[1+\frac{\lambda}{\delta}(1-I)\Bigr].
\label{p:eq:V}
\end{equation}
Its late-time value, the total expected output of one cell, is
\begin{equation}
V_\infty=\frac{a(1-b)}{a-1}=\frac{aB}{A}
=\frac{\lambda-\mu}{\delta}(1-b)+1,
\qquad
\mean{\Kb\mid\text{burst}}=\frac{V_\infty}{1-b}=\frac{a}{a-1},
\label{p:eq:Vinf}
\end{equation}
in which $V_\infty$ counts the internally cleared cells with their release
of zero and the conditional mean removes them. The integral of the survival
kernel, which anchors \cref{p:sec:projection}, is
\begin{equation}
\mean{T_{\rm prod}}=\int_0^\infty\Ifix(\alpha)\,\dd\alpha
=\frac{1}{\lambda}\log\frac{a}{a-1}
=\frac{1}{\lambda}\log\mean{\Kb\mid\text{burst}}.
\label{p:eq:lifetime}
\end{equation}

\subsection{Distribution-level facts}

Three results of \ChDist{} are used as black boxes. First, the burst-time
density is
\begin{equation}
-I'(t)=\delta J(t),
\label{p:eq:bursttime}
\end{equation}
defective, with total mass $1-b$. Second, the burst size, unconditional,
obeys
\begin{equation}
\pr{\Kb=k}=\frac{\delta}{\lambda}\,a^{-k},\qquad k\ge1,
\label{p:eq:burstlaw}
\end{equation}
with the mass $b$ at ``no burst''; conditioned on bursting this becomes
$(a-1)a^{-k}$, which is exactly the quasi-stationary distribution of the
load \cite{van2011quasi} --- both are geometric on $\{1,2,\ldots\}$ with ratio $1/a$. Third,
given that the burst occurs at time $t$, its size is a size-biased draw from
the state probabilities:
\begin{equation}
\mean{\Kb\mid\tau=t}=\frac{K(t)}{J(t)}
=1+\frac{2\lambda}{\delta}\bigl(1-I(t)\bigr),
\label{p:eq:sizebias}
\end{equation}
tending at late time to $(a+1)/(a-1)$, the size-biased quasi-stationary
mean.

\subsection{The second moment of the release}

The whole-host variance prediction of \cref{p:sec:open-problems} starts from
\begin{equation}
\mean{W_t^2}
=\frac{2(\lambda-\mu)}{\delta}V
-\frac{\lambda+\mu}{\delta}I
-K
+\frac{\lambda+\mu}{\delta}
+1,
\label{p:eq:EW2}
\end{equation}
with $\mathrm{Var}(W_t)=\mean{W_t^2}-V^2$. Its limit supplies
$\mathrm{Var}(\Kb)$, and hence the offspring variance
\cref{p:eq:offmoments}.

\subsection{Multiplicity of infection}

A cell founded by $k$ particles releases superlinearly more. Its moments and
release flux are
\begin{equation}
J_k(t)=kJ(t)I(t)^{k-1},\qquad
K_k(t)=kK(t)I(t)^{k-1}+k(k-1)J(t)^2I(t)^{k-2},
\label{p:eq:JkKk}
\end{equation}
\begin{equation}
g_k(t)=\delta K_k(t)
=\delta\lrb{kK I^{k-1}+k(k-1)J^2I^{k-2}},
\label{p:eq:gk}
\end{equation}
and its lifetime yield, for $k\ge2$,
\begin{equation}
V_\infty^{(k)}
=(1-b^k)
+\frac{2\lambda}{\delta}\Biggl[\frac{1}{k+1}-b^k+\frac{k\,b^{k+1}}{k+1}\Biggr]
+k(k-1)\frac{\lambda}{\delta}\int_1^b (x-a)(x-b)\,x^{k-2}\,\dd x,
\label{p:eq:Vk}
\end{equation}
simplifying to $V_\infty^{(k)}=k+\lambda/\delta$ when $\mu=0$. The
conditional mean burst $V_\infty^{(k)}/(1-b^k)$ increases with $k$: multiply
infected cells burst more often, and bigger. The pair
$\bigl(I(\alpha)^k-D(\alpha)^k,\,\delta K_k(\alpha)\bigr)$ is the MOI-$k$
row of \cref{p:tab:sg}; assumption (A4) of \cref{p:sec:assumptions} sets it
aside for the population layer.
