% ===========================================================================
%  tikz_style.tex --- shared TikZ conventions for this chapter's diagrams.
%
%  Reconstructed in Phase C.  The original lived at CH6 revise/style/ and is
%  absent from every copy of the tree; the three standalone `figure.tex` files
%  in figures/_work/ each `\input` it, and each defines its own boxes and
%  cards locally, so the only style they take from here is `bdc arrow`.
%
%  Conventions, per the note the diagrams carry: black Stealth arrows,
%  chapter-scale type, rounded state boxes, restrained grey fills.
% ===========================================================================

\tikzset{
  % the standard directed edge used throughout the chapter's diagrams
  bdc arrow/.style={
    -{Stealth[length=2.0mm,width=1.4mm]},
    line width=0.8pt,
    draw=black!80,
  },
  % a lighter variant for secondary flows
  bdc arrow faint/.style={
    bdc arrow,
    draw=black!45,
    line width=0.6pt,
  },
  % rounded state box, restrained grey fill
  bdc state/.style={
    draw=black!70,
    rounded corners=2pt,
    fill=black!3,
    inner sep=3pt,
    align=center,
    font=\footnotesize,
  },
  % absorbing state: the rupture state H is drawn with a double rule
  bdc absorbing/.style={
    bdc state,
    draw=black!85,
    line width=1.0pt,
    fill=black!7,
  },
  % rate label riding an edge
  bdc ratelabel/.style={
    font=\footnotesize,
    inner sep=1.5pt,
    fill=white,
    fill opacity=1,
    text opacity=1,
  },
  bdc note/.style={font=\scriptsize,text=black!62,align=center},
}
