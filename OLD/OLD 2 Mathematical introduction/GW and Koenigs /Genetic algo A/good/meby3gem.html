import numpy as np
import matplotlib.pyplot as plt
import re

expr = "PASTE_YOUR_EXPRESSION_HERE"

def replace_abs(expr):
    while '|' in expr:
        expr = re.sub(r'\|([^|]+)\|', r'abs(\1)', expr)
    return expr

expr = expr.replace("ln", "log")
expr = replace_abs(expr)

safe_namespace = {
    "x": None,
    "sin": np.sin,
    "cos": np.cos,
    "exp": np.exp,
    "log": np.log,
    "sqrt": np.sqrt,
    "abs": np.abs,
    "pi": np.pi,
}

x = np.linspace(0, 0.55, 1000)
safe_namespace["x"] = x

with np.errstate(all="ignore"):
    y = eval(expr, {"__builtins__": {}}, safe_namespace)
