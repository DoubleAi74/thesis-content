\section{Process definition}
\label{sec:process}
\SOURCE\
\texttt{3 BDC core/document MAIN/sections/04\_process\_definition.tex};
corrected statement also in
\texttt{new\_notes2/Core\_Publishable\_Work\_Summary.tex} \S``The single-cell process'',
\texttt{new\_notes/BMVR\_extension\_notes.tex} Part~I.

Starting from $\xt=1$ (one founder) unless stated, the process jumps
\begin{align*}
n &\to n+1 &&\text{at rate }\beta n,\\
n &\to n-1 &&\text{at rate }\mu n,\\
n &\to R &&\text{at rate }\delta n.
\end{align*}
On the jump to $R$ the cell bursts: $\wt$ is set to the pre-burst load and stays
there. Classical references: Brockwell--Gani--Resnick; Karlin--Tavar\'e.
What matters for applications is the closed-form family below.

\SOURCE\
joint process $(\xt,\wt)$: same file
\texttt{04\_process\_definition.tex} (``Post catastrophe count'');
before $\tau$, $\wt=0$; at catastrophe, $\wt\leftarrow \xt$ and $\xt\leftarrow R$.

\section{Root algebra}
\label{sec:roots}
\SOURCE\
\texttt{3 BDC core/document MAIN/sections/03\_reference\_values.tex},
\texttt{05\_analytical\_quantities.tex};
$A,B$ notation and $\Aa\Bb=\delta/\beta$ emphasised in
\texttt{new\_notes/BMVR\_extension\_notes.tex} Part~I.

\begin{equation}
\eta=\frac{\beta+\mu+\delta}{2\beta},\qquad
a=\eta+\sqrt{\eta^2-\tfrac{\mu}{\beta}},\qquad
b=\eta-\sqrt{\eta^2-\tfrac{\mu}{\beta}}.
\end{equation}
Write $\Aa:=a-1>0$, $\Bb:=1-b>0$, $\theta:=\beta(a-b)$.

\begin{result}[Vieta and the workhorse identity]
\begin{equation}
\label{eq:AB}
ab=\frac{\mu}{\beta},\qquad
a+b=\frac{\beta+\mu+\delta}{\beta},\qquad
\Aa\Bb=\frac{\delta}{\beta},\qquad
\Aa+\Bb=a-b.
\end{equation}
In particular $b<1<a$ always, and $\beta(1-a)(1-b)=-\delta$.
Also $\theta=\sqrt{(\beta+\mu+\delta)^2-4\beta\mu}$.
\end{result}

\NEW\ Identity $\Aa\Bb=\delta/\beta$ is elementary and should sit in every
reference table; it cancels prefactors throughout Ch.3.

\section{Three fixation functions}
\label{sec:fixation}
\SOURCE\
$I$ in Ch.3 \texttt{05\_analytical\_quantities.tex};
$D=p_0$ in \texttt{document MAIN/sections/05\_state\_probabilities.tex};
correct $\Ihat=I-D$ in
\texttt{new\_notes/BMVR\_extension\_notes.tex} Part~I
(thesis $\Ihat$ formula wrong for $\mu>0$).

\begin{definition}
\begin{align}
I(t)&=\Prob{\text{no burst by }t},\\
D(t)&=\Prob{\text{internal extinction by }t,\ \text{no burst}},\\
\Ihat(t)&=\Prob{\xt\notin\{0,R\}}=I(t)-D(t).
\end{align}
\end{definition}

Both $I$ and $D$ satisfy the Riccati equation
\begin{equation}
\label{eq:riccati}
f'=\beta(f-a)(f-b)=\mu-(\beta+\mu+\delta)f+\beta f^2,
\end{equation}
with $I(0)=1$ and $D(0)=0$. Writing $w:=\ee^{\theta t}\ge 1$:

\begin{result}[Closed forms for $I$, $D$, $\Ihat$]
\label{res:IDIhat}
\begin{empheq}[box=\fbox]{align}
I(t)&=\frac{a\Bb+b\Aa w}{\Bb+\Aa w},\\[0.25em]
D(t)&=\frac{ab\,(w-1)}{aw-b},\\[0.25em]
\Ihat(t)&=\frac{(a-b)^2\,w}{(\Bb+\Aa w)(aw-b)}.
\end{empheq}
Limits: $I(\infty)=D(\infty)=b$, $\Ihat(\infty)=0$, $\Ihat(0)=1$,
$\Ihat'(0)=-(\mu+\delta)$.
\end{result}

\begin{remark}[\COR\ thesis $\Ihat$ formula]
Ch.3 gives
$\Ihat(t)=\hat\eta\big/\bigl(1-(1-\hat\eta)\ee^{\hat\eta\beta t}\bigr)$ with
$\hat\eta=(\beta+\mu+\delta)/\beta$, from the wrong ODE
$\Ihat'=-(\beta+\mu+\delta)\Ihat+\beta\Ihat^2$.
The branching term on a birth must be $I^2-D^2$, not $\Ihat^2$.
Error is invisible when $\mu=0$ ($D\equiv 0$).
At $(\beta,\mu,\delta)=(1,0.2,0.05)$, $t=1$: correct $\Ihat\approx 0.8088$,
thesis formula $\approx 0.6675$.
Ch.4 state probabilities $p_n$ sum over $n\ge 1$ to the correct $\Ihat$.
\end{remark}

\begin{result}[\COR\ multiplicity of infection --- survival]
For $X_0=k$ independent founders (one cell, linear rates, shared catastrophe):
\begin{equation}
I_k(t)=I(t)^k,\qquad
D_k(t)=D(t)^k,\qquad
\Ihat_k(t)=I(t)^k-D(t)^k.
\end{equation}
Thesis claim $\Ihat_k=\Ihat^{\,k}$ is false.
\end{result}

\section{Moments and mean release}
\label{sec:moments}
\SOURCE\
\texttt{3 BDC core/document MAIN/sections/05\_analytical\_quantities.tex},
\texttt{07\_expected\_number\_of\_units\_j\_t.tex},
\texttt{08\_variance\_and\_standard\_deviation\_of.tex}
(and duplicate \texttt{10\_\ldots});
simplifications/corrections in
\texttt{new\_notes/BMVR\_extension\_notes.tex} Part~I,
\texttt{new\_notes2/Core\_Publishable\_Work\_Summary.tex} \S moments.

From $I'=-\delta J$ and the Riccati:

\begin{result}[First and second moments; mean release]
\begin{empheq}[box=\fbox]{align}
J(t)&=\E[\xt]
=\frac{(a-b)^2\,w}{(\Bb+\Aa w)^2}
=-\frac{\beta}{\delta}(I-a)(I-b),
\label{eq:J}\\[0.35em]
K(t)&=\E[\xt^2]
=\Bigl[1+\tfrac{2\beta}{\delta}(1-I)\Bigr]J
=\frac{2\beta}{\delta}(\kappa-I)J,
\quad \kappa=1+\frac{\delta}{2\beta},
\label{eq:K}\\[0.35em]
V(t)&=\E[\wt]
=(1-I)\Bigl[1+\tfrac{\beta}{\delta}(1-I)\Bigr].
\label{eq:V}
\end{empheq}
\end{result}

\begin{remark}[Simplifications]
\begin{itemize}
\item Thesis $J$ carries a cluttered prefactor; $\Aa\Bb=\delta/\beta$ cancels it
  and $J(0)=1$ is manifest.
\item Thesis $V=\frac{\beta-\mu}{\delta}(1-I)-J+1$ equals \eqref{eq:V}, a
  \textbf{quadratic in $(1-I)$ alone} --- no separate $J$ or $\mu$.
\item Consequently $V'=\delta K$, so $g:=\delta K$ is the release flux.
\item Alternative: $V=\frac{\beta}{\delta}\bigl[(I-\kappa)^2-(1-\kappa)^2\bigr]$.
\end{itemize}
\end{remark}

\begin{result}[Lifetime-time release and conditional mean burst size]
\begin{equation}
V_\infty=\frac{a(1-b)}{a-1}=\frac{a\Bb}{\Aa}
=\frac{\beta-\mu}{\delta}(1-b)+1,
\qquad
\E[\Kb\mid\text{burst}]=\frac{V_\infty}{1-b}=\frac{a}{a-1}.
\end{equation}
$V_\infty$ \emph{includes} realisations that clear internally and release nothing
(mass $b$ at $0$). The conditional mean removes them.
When $\mu=0$: $b=0$, $a=1+\delta/\beta$, $V_\infty=1+\beta/\delta=\langle X\rangle_{\rm QS}$.
When $\mu>0$ they differ (e.g.\ $(1,0.2,0.05)$: $17.23$ vs $13.99$).
\end{result}

\section{Second moment of release \COR}
\label{sec:EW2}
\SOURCE\
raw (wrong signs):
\texttt{3 BDC core/document MAIN/sections/08\_variance\_and\_standard\_deviation\_of.tex};
corrected formula:
\texttt{new\_notes/BMVR\_extension\_notes.tex} Part~I \S``Second moment of the release''.

Starting from $\dd\E[\wt^2]/\dd t=\delta\E[\xt^3]$ and eliminating the third
moment via the $K$ ODE, with correct initial data
$(I,J,K,V,\E[W^2])(0)=(1,1,\mathbf{1},0,0)$:

\begin{result}[Second moment of cumulative release]
\begin{empheq}[box=\fbox]{equation}
\E[\wt^2]
=\frac{2(\beta-\mu)}{\delta}V
-\frac{\beta+\mu}{\delta}I
-K
+\frac{\beta+\mu}{\delta}
+1.
\label{eq:EW2}
\end{empheq}
$\mathrm{Var}(\wt)=\E[\wt^2]-V^2$.
\end{result}

\begin{remark}[\COR\ thesis errors]
Thesis had wrong signs on $I$ and $K$, and omitted $+1$ (from $K(0)=1$, not $0$).
Cause: prose used $J=+\delta^{-1}I'$ instead of $J=-\delta^{-1}I'$.
Independent check $\mu=0$, $t\to\infty$: geometric on $\{1,2,\ldots\}$ with
success $s=\delta/(\beta+\delta)$ gives
$\E[W^2]=(2-s)/s^2=2\beta^2/\delta^2+3\beta/\delta+1$, matching \eqref{eq:EW2}.
\end{remark}

\begin{remark}[\COR\ $K$ as function of $I$]
Correct form (Ch.3 \S11, not \S9):
$K=\frac{2\beta}{\delta}(\kappa-I)J$ with $J=-\frac{\beta}{\delta}(I-a)(I-b)$.
The \S9 version has wrong power of $\delta$ and a sign error. Prefer the
$(I,J)$ form \eqref{eq:K} over expanded polynomials.
\end{remark}

\begin{result}

\[
\varphi(t)=-I'(t)=\delta J(t),\qquad
\int_0^\infty\varphi(t)\,\dd t=1-b.

\section{State probabilities are geometric at every time}
\label{sec:pn}
\SOURCE\
\texttt{document MAIN/sections/05\_state\_probabilities.tex};
QSD discussion:
\texttt{document MAIN/sections/06\_quasi\_stationary\_distribution.tex}.

For $n\ge 1$,
\begin{equation}
\label{eq:pn}
p_n(t)=\Bigl(\frac{a-b}{b-aw}\Bigr)^{\!2}w\;P(t)^{\,n-1},
\qquad P(t)=\frac{1-w}{b-aw},\qquad w=\ee^{\theta t}.
\end{equation}
Conditional on $\xt\ge 1$, the load is geometric with ratio $P(t)$ at every $t$,
and $P(t)\to 1/a$.

\section{Burst size equals the quasi-stationary distribution}
\label{sec:qsd}
\NEW\ ingredients in
\texttt{document MAIN/sections/05\_state\_probabilities.tex},
\texttt{06\_quasi\_stationary\_distribution.tex},
\texttt{07\_burst\_size\_distribution.tex};
identity stated in
\texttt{new\_notes/BMVR\_extension\_notes.tex},
\texttt{new\_notes2/Core\_Publishable\_Work\_Summary.tex}.

\begin{result}[Burst size $=$ QSD]
Both are geometric on $\{1,2,\ldots\}$ with ratio $1/a$:
\begin{align}
\mathrm{QSD}(n)
&=\lim_{t\to\infty}\Prob{\xt=n\mid \xt\notin\{0,R\}}
=(a-1)\,a^{-n},\\
\Prob{\Kb=k\mid\text{burst}}
&=(a-1)\,a^{-k}.
\end{align}
Unconditionally (including no-burst realisations),
$\Prob{\Kb=k}=(\delta/\beta)\,a^{-k}$ for $k\ge 1$, with mass $b$ at $0$.
\end{result}

\begin{result}[Proof of the burst-size law --- no simulation required]
With $P(t)=(1-w)/(b-aw)$,
\[
\frac{\dd}{\dd t}\Bigl[\frac{P^k}{k\beta}\Bigr]=p_k(t).
\]
Hence
\[
\Prob{\Kb=k}
=\delta k\int_0^\infty p_k(t)\,\dd t
=\frac{\delta}{\beta}\bigl[P^k\bigr]_0^\infty
=\frac{\delta}{\beta}\,a^{-k}.
\]
Checks: (i) $\sum_{k\ge 1}(\delta/\beta)a^{-k}=1-b$;
(ii) $\sum_{k\ge 1}k(\delta/\beta)a^{-k}=V_\infty$.
\end{result}

Immediate QSD moments:
\begin{equation}
\langle X\rangle_{\rm QS}=\frac{a}{a-1},\qquad
\mathrm{Var}_{\rm QS}(X)=\frac{a}{(a-1)^2},\qquad
\langle X^2\rangle_{\rm QS}=\frac{a(a+1)}{(a-1)^2}.
\end{equation}

\begin{remark}[\OPEN\ conceptual proof]
Algebra is complete. A structural argument (size-biasing composed with burst
intensity, or Yaglom-type limit) would make the identity quotable rather than
coincidental.
\end{remark}

\section{Size-biasing and late bursts}

A cell bursts at rate $\delta\xt$, so given burst at time $t$ the size is
size-biased:
\[
\Prob{\Kb=k\mid\tau=t}=\frac{k\,p_k(t)}{J(t)},\qquad
\E[\Kb\mid\tau=t]=\frac{K(t)}{J(t)}
\;\xrightarrow[t\to\infty]\;
1+\frac{2\beta}{\delta}(1-b).
\]
A very late burst is roughly twice the size of an average one.

\section{Resolving Ch.4 \S8 ``curious circularity''}
\COR\
\SOURCE\
\texttt{document MAIN/sections/08\_further\_considerations.tex};
resolution in
\texttt{new\_notes/BMVR\_extension\_notes.tex} Part~I.

Ch.4 computes $\E[\Kb\mid\tau>t]=(\E[\Kb]-V(t))/I(t)$, finds the $t\to\infty$
limit returns $\E[\Kb]$, and records confusion.

With consistent definitions:
\begin{equation}
\E\bigl[\Kb\mathbf 1\{\text{burst}\}\mid\tau>t\bigr]
=\frac{V_\infty-V(t)}{I(t)}
\;\xrightarrow{}\;0,
\end{equation}
correct: conditioning on ``has not burst yet'' converges to ``never bursts''.

The interesting limit conditions on eventual burst:
\begin{equation}
\E[\Kb\mid \tau>t,\ \text{burst eventually}]
=\frac{V_\infty-V(t)}{I(t)-b}
\;\xrightarrow{}\;
\frac{\langle X^2\rangle_{\rm QS}}{\langle X\rangle_{\rm QS}},
\end{equation}
by l'H\^opital ($V'=\delta K$, $I'=-\delta J$).

K_k(a)&=k\,K(a)\,I(a)^{k-1}+k(k-1)J(a)^2\,I(a)^{k-2},\\[0.25em]
g_k(a)&=\delta K_k(a).
\end{empheq}
\end{result}

\begin{remark}[\COR\ what was wrong]
Earlier notes wrote $g_k=\delta[kK+k(k-1)J^2]$ (free sum of independent
counts). That omits the $I$-powers from the shared survival indicator.
Master-equation check at $(1,0.2,0.05)$, $t=1$, $k=2$: wrong $K_2\approx 23.39$;

\end{remark}

\begin{result}[Lifetime mean release from $k$ founders]
\begin{align}
V_\infty^{(1)}&=V_\infty=\frac{a(1-b)}{a-1},\\[0.3em]
V_\infty^{(k)}
&=(1-b^k)
+\frac{2\beta}{\delta}\Biggl[\frac{1}{k+1}-b^k+\frac{k\,b^{k+1}}{k+1}\Biggr]
\nonumber\\
&\quad
+k(k-1)\frac{\beta}{\delta}\int_1^b (x-a)(x-b)\,x^{k-2}\,\dd x

\label{eq:Vk}
\end{align}

\begin{equation}

\end{equation}
(e.g.\ $\beta=1,\delta=0.1$: $11,12,13,14,15,\ldots$).

Lifetime yield is \textbf{not} $k V_\infty$ (shared first-catastrophe dumps the
whole cell once). Conditional mean given eventual burst \emph{increases} with $k$.
\end{result}

\section{Chained immediate-transfer model ($N=2$ and general $M$)}
\label{sec:chain}
\SOURCE\
\texttt{N=2 Immediate Transfer content/N\_eq\_2.tex}
(PDF: \texttt{N=2 Immediate Transfer content/N\_eq\_2.pdf}).

\paragraph{Setup.}
$M$ host cells. One starts with a single particle. On rupture, the entire load
is transferred immediately into a new empty cell (no extracellular phase).
Process continues until $M$ ruptures. Extreme caricature of serial infection;
useful for tractable multi-cell burst statistics.

\paragraph{Notation.}
$r_i$ = load at the $i$th rupture; $t_i$ = time of $i$th rupture;
$T_i=t_i-t_{i-1}$ = inter-rupture interval. Birth rate $\lambda$ in source
text $=$ our $\beta$.

\paragraph{Rupture-size laws ($\mu=0$).}
With $s=\delta/(\delta+\beta)$ and $\rho=\beta/(\delta+\beta)$,
\begin{equation}
\Prob{r_k=n}
=\mathcal{A}^{(n)}_{k}\, s^{k}\,\rho^{n-1},
\qquad
\mathcal{A}^{(n)}_{k}
=\frac{(n+k-2)!}{(k-1)!\,(n-1)!}
=\binom{n+k-2}{k-1}.
\end{equation}
Special cases:
\begin{align*}
\Prob{r_1=n}&=s\,\rho^{n-1}
&&\text{(geometric --- matches one-cell BDC, $\mu=0$)},\\
\Prob{r_2=n}&=n\,s^2\,\rho^{n-1},\\
\Prob{r_3=n}&=\tfrac{n(n+1)}{2}s^3\,\rho^{n-1}.
\end{align*}
Recurrence: $\mathcal{A}^{(n)}_{k}=\sum_{i=1}^{n}\mathcal{A}^{(i)}_{k-1}$ with
$\mathcal{A}^{(i)}_{1}=1$.

\paragraph{First rupture time from $m$ founders.}
If $S(t)=I(t)$ is the no-burst probability from one founder, then
\[
\Prob{t_1>t}=S(t)^m,\qquad
\text{density }\mathcal{F}_m(t)=-m S(t)^{m-1}S'(t).
\]

general $\mu$; link of chained $r_k$ laws to MOI kernel $g_k$ under immediate
transfer.

\section{Why constant-$p$ substitution fails}

Ch.4 proposed replacing BMVR release rate $p$ by $\langle X\rangle_{\rm QS}$.

\begin{framed}
\textbf{Obstruction is dimensional.} $p$ has units particles/time;
$\langle X\rangle_{\rm QS}$ is a pure count. When $\mu=0$,
$\langle X\rangle_{\rm QS}=V_\infty$, so the proposal substitutes total lifetime
output for a rate.

Candidate integrals also fail:
\begin{itemize}
\item $\int\varphi(t)\,\E[\Kb\mid\tau=t]\,\dd t=V_\infty$ --- a count, not a rate.
\item Weighting by the burst-time density is wrong; the correct weight is the
  \textbf{age distribution of currently infected cells}.
\end{itemize}
Further: $\Vfree(t)=N_0 V(t)$ only for synchronously infected cells. Asynchrony
is the whole problem --- and the whole solution (renewal).
\end{framed}

\begin{result}[New viral dynamics --- burst-aware BMVR]
\label{res:new-vde}
With incidence $i(t)=\gamma T\,\Vfree(t)$,
\begin{empheq}[box=\fbox]{align}
\Icell(t)
&= \Icell_0\,\Ihat(t)
+\int_0^t i(t-\alpha)\,\Ihat(\alpha)\,\dd\alpha

\label{eq:Icell-full}\\[0.4em]

&= \Icell_0\,g(t)
+\int_0^t i(t-\alpha)\,g(\alpha)\,\dd\alpha

\label{eq:Vfree-full}
\end{empheq}
\end{result}

\begin{center}

\end{tabular}
\end{center}

the consequences below ($p_{\rm eff}$, flooding, etc.).
\end{remark}

=i+\Icell_0\Ihat'+i*\Ihat'.

No constant $d_{\Icell}$ reproduces this for all histories of $i$; reduction to
classical form holds only for pure exponentials (next section).

$G_{\rm off}(z)=z$ is quadratic in $y$; the non-trivial root is
\begin{equation}
z_{\rm ext}^{\rm burst}
=\frac{a(1-q+qb)-1+q}{q}
=\frac{a-1}{q}+1-a(1-b).
\end{equation}

\section{Flooding advantage --- main theorem}
\label{sec:flooding}
\SOURCE\
\texttt{new\_notes/BMVR\_extension\_notes.tex}
Part~II \S``The flooding advantage, answered exactly'';
\texttt{new\_notes2/Core\_Publishable\_Work\_Summary.tex}
\S``The flooding advantage'';
$L=1$ law-match and $r$-trade-off strengthened in this document.

Comparator: BMVR cell with $\mathrm{Exp}(d_{\Icell})$ lifetime, constant rate
$p$, matched so $p/d_{\Icell}=V_\infty$ and same $q$. Offspring is geometric with
mean $m=q V_\infty$, so $z_{\rm ext}^{\rm bud}=1/m$ when $m>1$.

Write $L:=a(1-b)=a-\mu/\beta$, so $V_\infty=L/(a-1)$.

\begin{result}[Flooding advantage]
\label{res:flood}
For $m>1$,
\begin{equation}
\label{eq:flood}
z_{\rm ext}^{\rm burst}-z_{\rm ext}^{\rm bud}
=\bigl(L-1\bigr)\Bigl(\frac{1}{m}-1\Bigr).
\end{equation}
Hence bursting strictly lowers extinction probability relative to matched-mean
budding iff $L>1$, i.e.
\begin{equation}
\boxed{\;
\frac{1}{\delta}\;<\;\frac{1}{\beta}+\frac{1}{\mu}\;}
\quad
\bigl(\text{when }\mu=0\text{: always, for all }\delta>0\bigr).
\end{equation}
When $L=1$ the extinction probabilities are identical for every $q$.
\end{result}

\begin{result}[\NEW\ strengthening at $L=1$]
When $L=1$ (equivalently $b=\delta/\beta$), the offspring law is
\textbf{exactly geometric} with mean $m$ --- not merely same extinction
probability. Hence variances match as well.
Equivalence chain:
\[
L=1
\;\Longleftrightarrow\;
a=1+\frac{\mu}{\beta}
\;\Longleftrightarrow\;
\delta(\beta+\mu)=\beta\mu
\;\Longleftrightarrow\;
\frac{1}{\delta}=\frac{1}{\beta}+\frac{1}{\mu}.
\]
\end{result}

\begin{result}[\NEW\ variance ordering]
At matched mean $m$, $\mathrm{Var}_{\rm burst}<\mathrm{Var}_{\rm bud}$ iff
$L>1$, with equality at $L=1$ (geometric case), and the reverse when $L<1$.
\end{result}

\paragraph{Interpretation.}
Bursting concentrates output (lower offspring variance when $L>1$) but carries
all-or-nothing risk of internal clearance (probability $b$). Advantage survives
when the burst timescale $1/\delta$ is shorter than the harmonic-type combination
of birth and death timescales.

\CAVEAT\ Comparison holds the offspring \emph{mean} fixed --- branching
structure only. Nonlinear effects (immune saturation, local depletion, Allee)
are separate; name both mechanisms in any write-up (Komarova-type discussion).

\section{Deterministic growth-rate trade-off \NEW}
\label{sec:r-tradeoff}
\SOURCE\
numerical comparison developed for this document
(not in Ch.3/4; related discussion of $R_0$ vs $r$ in
\texttt{new\_notes/BMVR\_extension\_notes.tex} Part~II).

At matched $R_0^{\mathrm{ODE}}$ (or matched $m$) \textbf{and} matched mean
productive lifetime $d=1/\E[T_{\rm prod}]$, $p=V_\infty d$:

\begin{result}[Budding is faster in the exponential phase]
In all regimes checked ($L>1$, $L=1$, $L<1$),
\[
r_{\mathrm{bud}} > r_{\mathrm{burst}}.
\]
Example: $(\beta,\mu,\delta)=(1,0,0.1)$, $R_0=2$:
$r_{\mathrm{burst}}\approx 0.18$, $r_{\mathrm{bud}}\approx 0.25$.
\end{result}

\textbf{Biological headline for a paper:} at fixed total output per cell,
bursting can \emph{lower early extinction} (when $L>1$) while \emph{slowing
deterministic invasion speed} (generation-time delay). $R_0$ fixed; $z_{\rm ext}$
and $r$ can move in opposite directions.

\section{Optimal burst rate at fixed mean offspring \NEW}

At fixed $m=R_0$ (adjust $q=m/V_\infty$) and $\mu=0$: faster bursting
(larger $\delta$) lowers $z_{\rm ext}$ until $V_\infty$ becomes too small to
support the target $m$ with $q\le 1$.

At fixed $m$ and $\mu>0$: bursting beats budding only once $\delta$ crosses the
$L=1$ threshold $\delta_*=\beta\mu/(\beta+\mu)$; below that threshold, budding
wins on extinction.

\section{Five modular ODE cases (mean-field ladder)}
\label{sec:cases}

\S``Building BMVR piece by piece: five cases''.

\paragraph{Case 1 --- Classic BMVR.}
\[
\Icell'=\gamma T\Vfree-d_{\Icell}\Icell,\qquad
\Vfree'=p\Icell-c\Vfree.
\qquad
R_0=\frac{\gamma T\,p}{c\,d_{\Icell}}.
\]

\paragraph{Case 2 --- Eclipse + classic release.}
\[
E'=\gamma T\Vfree-(a+d_E)E,\quad
\Icell'=aE-d_{\Icell}\Icell,\quad
\Vfree'=p\Icell-c\Vfree.
\]
$R_0=\frac{\gamma T}{c}\cdot\frac{a}{a+d_E}\cdot\frac{p}{d_{\Icell}}$.
Multi-stage (Erlang) eclipse: chain $E_1\to\cdots\to E_n\to\Icell$.

\paragraph{Case 3 --- Intracellular stock + continuous export
(mean-linear accumulation).}
\[
\Icell'=\gamma T\Vfree-d_{\Icell}\Icell,\quad
Q'=\nu\Icell-\varepsilon Q-d_{\Icell}Q,\quad
\Vfree'=\varepsilon Q-c\Vfree.
\]
Quasi-steady stock: $p_{\mathrm{eff}}=\varepsilon\nu/(\varepsilon+d_{\Icell})$;
Case~1 is the fast-export limit. Prefer immigration-type $\nu$ for HIV-like
biology (Section~\ref{sec:hiv-growth}).

\paragraph{Case 4 --- Stock + load-proportional full dump (mean-field reset).}
Dump flux $\propto Q^2/\Icell$ (from $\delta\E[X^2]$ with $X\sim$ load per cell
closure $Q/\Icell$). Recovers constant-$p$ only in mature-cell limit.

\paragraph{Case 5 --- Eclipse + stock + proportional dump.}
Incidence $\to E\to\Icell$; only productive cells accumulate $Q$ and dump.

\section{Reset-catastrophe process (intermediate between budding and BDC)}
\label{sec:reset}

\section{One-page story (for introductions)}

Many within-host models treat infected cells as black boxes releasing virions at
constant rate $p$ (BMVR). Lytic pathogens grow inside a cell and dump their load
in a single burst. Model one cell as a classical BDC process; derive closed forms
for survival, load, release, burst timing and size; embed those formulas in a
renewal BMVR so the population model is forced by the intracellular process.

At fixed total output per cell, $R_0$ is unchanged by bursting, but generation
times change. Bursting lowers establishment extinction relative to matched-mean
budding if and only if $1/\delta<1/\beta+1/\mu$ (always when $\mu=0$); at the
boundary the offspring laws coincide. Deterministic growth is typically faster
under budding. Effective release rate $p_{\rm eff}$ falls as epidemic growth rate
rises (young cells dominate). Three intracellular rates cannot be recovered from
two BMVR parameters alone.

The cleanest single-cell fact: the burst-size distribution is exactly the
quasi-stationary load distribution --- both geometric with ratio $1/a$.

\paragraph{$\mu=0$.}
$b=0$, $a=1+\delta/\beta$, $D\equiv 0$, $\Ihat=I$,

\paragraph{$L=1$ boundary.}
$\delta=\beta\mu/(\beta+\mu)$, $b=\delta/\beta$, offspring law geometric,
$z_{\rm ext}^{\rm burst}=z_{\rm ext}^{\rm bud}$, equal variances.

\paragraph{Young-cell / fast-growth limit.}
$p_{\rm eff}\to\delta$, $d_{\rm eff}\to\mu+\delta$.

\paragraph{Set-point / $r=0$ matching.}
$p_{\rm eff}=V_\infty/\E[T_{\rm prod}]$, $d_{\rm eff}=1/\E[T_{\rm prod}]$.
