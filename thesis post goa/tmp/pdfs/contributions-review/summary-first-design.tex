\oimapcard{03}{One bursting cell}
  {\oimapchapter{bdc:ch:core}{Chapters~\ref*{bdc:ch:core} and~\ref*{dist:ch:distribution-theory}}}
  {Expresses load moments using one no-catastrophe probability and tracks
   release at finite time. Derives burst timing, explains why the burst-size and
   quasi-stationary laws agree, and extends the calculation to several
   founders and successive cell-to-cell transfers.}%
\hfill
\oimapcard{04}{From cells to infection}
  {\oimapchapter{p:ch:one-cell-to-population}{Chapter~\ref*{p:ch:one-cell-to-population}}}
  {An age-structured infection model with survival and release kernels
   derived from the cell process. It shows why fitted constant rates
   depend on the growth phase, and why one exponential phase cannot
   identify all three intracellular rates.}
\par\vspace{12pt}

\oimapcard{05}{A branching constant explained}
  {\oimapchapter{gw:ch:galton-watson}{Chapter~\ref*{gw:ch:galton-watson}}}
  {Kolmogorov's survival constant $A(p)$ is linked to the logistic map's
   Koenigs function. A hypertranscendence proof rules out an elementary
   formula for that function in its spatial argument; the closed-form
   question for $A(p)$ remains open.}%
\hfill
\oimapcard{06}{Evolution under changing rates}
  {\oimapchapter{var:ch:rates}{Chapter~\ref*{var:ch:rates}}}
  {A speciation model driven by unused population capacity yields an
   explicit limiting species-count distribution and variance. A moving
   critical point is derived for the Pimentel resistance model, locating
   reversals of competitive advantage.}
\par
